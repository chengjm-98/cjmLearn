# 代码提交

# 打包/部署/镜像/容器配置/发到服务器

```jsx
源码 (React/Vue)
   │
   ▼
npm install + build
   │
   ▼
生成静态文件 dist/
   │
   ▼
Docker build (node + nginx)
   │
   ▼
Docker 镜像 my-frontend:latest
   │
   ▼
上传服务器 / 私有仓库
   │
   ▼
docker run -d -p 80:80 my-frontend:latest
   │
   ▼
访问网站

```

# 发布

```jsx
清理 Jenkins 工作区

根据环境克隆不同分支的 Git 代码

构建 Docker 镜像并推送到私有仓库

在目标服务器上部署容器并挂载 Nginx 配置

输出构建报告和访问 URL
```

```jsx
#!groovy
// 导入 Jenkins 插件中用于管理字符串类型凭证的类（这里没有实际使用）
import org.jenkinsci.plugins.plaincredentials.StringCredentials

// 定义 Git 仓库地址
repoURL = "ssh://vcssh@phabricator.intern.yuansuan.cn/diffusion/62/cae.git"

// Jenkins 工作空间路径（用来 clone 代码）
repoPath = "/opt/jenkins/workspace/gw-ca-full-fe"

// 前端子项目工作目录路径
Workdir = "/opt/jenkins/workspace/gw-ca-full-fe/ca-full/frontend"

pipeline {
    // 指定 Jenkins Agent 节点标签
    agent {label 'tstage'}

    stages {

        // ==========================
        stage('clean') {
        // 清理工作区
            steps {
                sh"""
                    rm -rf ${repoPath}       # 删除之前的代码目录
                    mkdir -p ${repoPath}     # 新建干净的目录
                """
            }
        }

        // ==========================
        stage('clone') {
        // 克隆 Git 仓库代码
            steps {
                script {
                    if("${envs}" == "dev") {
                        // 如果是开发环境，克隆最新代码（只取最近一次提交）
                        sh"""
                            git clone --depth 1 ${repoURL} ${repoPath}
                        """
                    } else if("${envs}" == "stage") {
                        // 如果是预发布环境，克隆指定分支 (ca-full-BUILD_NUMBER)
                        sh"""
                            git clone --branch ca-full-${BUILD_NUMBER} --depth 1 ${repoURL} ${repoPath}
                        """
                    }
                }
            }
        }

        // ==========================
        stage('build') {
        // 前端构建 & Docker 镜像打包 & 推送
            steps {
                echo "版本号:${BUILD_NUMBER}"  // 打印 Jenkins 构建号
                echo '开始打包.'
                sh"""
                cd ${Workdir}  # 进入前端项目目录
                docker login -u admin -p yskj2407 1.117.192.82:8666   # 登录私有 Docker 仓库
                docker build -t 1.117.192.82:8666/ca-full/ca-full-fe:${BUILD_NUMBER} .  # 构建 Docker 镜像并打上 tag
                docker push 1.117.192.82:8666/ca-full/ca-full-fe:${BUILD_NUMBER}         # 推送镜像到仓库
                """
                echo '打包完成.'

                script {
                    // 创建部署目录
                    if("${envs}" == "dev") {
                        sh "ssh tdev \"mkdir -p /opt/ca-full/front\""
                    } else if("${envs}" == "stage") {
                        sh "mkdir -p /opt/ca-full/front"
                    }
                }
                echo 'build end.'
            }
        }

        // ==========================
        stage('deploy') {
        // 部署前端 Docker 容器并拷贝 Nginx 配置
            steps {
                echo 'deploy start.'
                script {
                    if("${envs}" == "dev") {
                        // ==========================
                        // 开发环境部署（远程 tdev 服务器）
                        sh "scp  ${Workdir}/nginx/fe_tdev.conf tdev:/opt/ca-full/front/"  // 复制 Nginx 前端配置
                        sh "scp  ${Workdir}/nginx/nginx.conf tdev:/opt/ca-full/front/"    // 复制 Nginx 主配置
                        sh "ssh tdev \"docker rm -f ca-full-fe || true\""                  // 删除旧容器
                        sh "ssh tdev \"docker image prune -a -f || true\""                 // 清理旧镜像
                        sh "ssh tdev \"docker login -u admin -p yskj2407 1.117.192.82:8666\"" // 登录 Docker 仓库
                        sh "ssh tdev \"docker pull 1.117.192.82:8666/ca-full/ca-full-fe:${BUILD_NUMBER}\"" // 拉取最新镜像
                        sh "ssh tdev \"docker run -itd -p 8056:8056 -v /opt/ca-full/front/nginx.conf:/etc/nginx/nginx.conf -v /opt/ca-full/front/fe_tdev.conf:/etc/nginx/conf.d/default.conf  --restart=always  --name ca-full-fe 1.117.192.82:8666/ca-full/ca-full-fe:${BUILD_NUMBER}\""
                        // 启动容器并挂载 Nginx 配置文件，端口映射 8056
                    } else if("${envs}" == "stage") {
                        // ==========================
                        // 预发布环境部署（本机 Docker）
                        sh "cp -r ${Workdir}/nginx/fe_tstage.conf /opt/ca-full/front/"  // 复制 Nginx 前端配置
                        sh "cp -r ${Workdir}/nginx/nginx.conf /opt/ca-full/front/"      // 复制 Nginx 主配置
                        sh "docker rm -f ca-full-fe || true"                             // 删除旧容器
                        sh "docker login -u admin -p yskj2407 1.117.192.82:8666"        // 登录 Docker 仓库
                        sh "docker pull 1.117.192.82:8666/ca-full/ca-full-fe:${BUILD_NUMBER}"  // 拉取最新镜像
                        sh "docker run -itd  -p 8056:8056 -v /opt/ca-full/front/nginx.conf:/etc/nginx/nginx.conf -v /opt/ca-full/front/fe_tstage.conf:/etc/nginx/conf.d/default.conf  --restart=always --name ca-full-fe 1.117.192.82:8666/ca-full/ca-full-fe:${BUILD_NUMBER}"
                        // 启动容器并挂载 Nginx 配置文件，端口映射 8056
                    }
                }
                echo 'deploy end.'
            }
        }

        // ==========================
        stage('report') {
        // 构建结果报告
            steps {
                echo 'Build && Test succeeded.'
                echo "http://10.0.4.66:8056/?ticket=569619c5-6100-4f35-a5fb-43fc2ade667f"  // 开发环境 URL
                echo "http://10.0.4.67:8056/?ticket=569619c5-6100-4f35-a5fb-43fc2ade667f"  // 预发布环境 URL
                echo "${BUILD_NUMBER}"  // 打印构建号
            }
        }
    }
}
```
