# 使用id

给每个请求打上顺序编号，返回时只处理最新编号的请求结果。

```jsx
let latestRequestId = 0;

input.oninput = async (e) => {
  const requestId = ++latestRequestId; // 为每次请求生成唯一 id
  const result = await fetch(`/search?q=${e.target.value}`).then((res) =>
    res.json(),
  );

  if (requestId === latestRequestId) {
    // 只渲染最新请求的结果
    render(result);
  }
};
```

| 请求 | requestId | latestRequestId 当时的值 |
| ---- | --------- | ------------------------ |
| A    | 1         | 1                        |
| B    | 2         | 2                        |
| C    | 3         | 3                        |

## 解释

```jsx
请求 A 返回时，requestId = 1，但此时 latestRequestId = 3
→ 条件不成立 → 不渲染

请求 C 返回时，requestId = 3，此时 latestRequestId = 3
→ 条件成立 → 渲染结果
```

# 使用react封装

```jsx
const {
  data,
  loading,
  error,
  run,
} = useLatestRequest(apiFn, {
  debounce: 300,   // 或 throttle: 300
});
import { useRef, useState, useCallback, useEffect } from "react";

type Options = {
  debounce?: number;
  throttle?: number;
};


export function useLatestRequest<T, P extends any[]>(
  asyncFn: (...args: P) => Promise<T>,
  options: Options = {}
) {
  const { debounce: debounceDelay, throttle: throttleDelay } = options;

  const latestRequestIdRef = useRef(0);
  const mountedRef = useRef(true);
  const runRef = useRef<((...args: P) => void) | null>(null);

  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<any>(null);

  useEffect(() => {
    return () => {
      mountedRef.current = false;
    };
  }, []);

  const _run = useCallback(
    async (...args: P) => {
      const requestId = ++latestRequestIdRef.current;

      if (mountedRef.current) {
        setLoading(true);
        setError(null);
      }

      try {
        const result = await asyncFn(...args);

        if (
          mountedRef.current &&
          requestId === latestRequestIdRef.current
        ) {
          setData(result);
          setLoading(false);
        }

        return result;
      } catch (err) {
        if (
          mountedRef.current &&
          requestId === latestRequestIdRef.current
        ) {
          setError(err);
          setLoading(false);
        }
        throw err;
      }
    },
    [asyncFn]
  );

  // ⭐ 包装 debounce / throttle
  if (!runRef.current) {
    let wrapped = _run;

    if (debounceDelay) {
      wrapped = debounce(_run, debounceDelay);
    } else if (throttleDelay) {
      wrapped = throttle(_run, throttleDelay);
    }

    runRef.current = wrapped;
  }

  return {
    data,
    loading,
    error,
    run: runRef.current!,
  };
}

```

## 具体使用

```jsx
const searchApi = (kw: string) =>
  fetch(`/search?q=${kw}`).then(res => res.json());

function Search() {
  const [keyword, setKeyword] = useState("");

  const {
    data,
    loading,
    run,
  } = useLatestRequest(searchApi, {
    debounce: 300,
  });

  useEffect(() => {
    if (!keyword) return;
    run(keyword);
  }, [keyword, run]);

  return (
    <>
      <input
        value={keyword}
        onChange={e => setKeyword(e.target.value)}
      />

      {loading && <p>loading...</p>}

      {data?.map((item: any) => (
        <div key={item.id}>{item.name}</div>
      ))}
    </>
  );
}

```
